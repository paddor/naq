# With the system account: /accountz names it, and service exports show
# their latency samples. Every account imports the services of the
# system account, which exports them.
#! accounts {
#!   A {
#!     users [{user: a, password: p}]
#!     exports [
#!       {service: svc.>, latency: {sampling: 50, subject: lat.a}}
#!       {service: hdr.>, latency: {sampling: headers, subject: lat.h}}
#!     ]
#!   }
#!   B { imports [{service: {account: A, subject: svc.x}}] }
#! }
#= accounts
#= exports
#= imports
