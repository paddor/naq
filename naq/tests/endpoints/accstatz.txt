# /accstatz counts the connections and subscriptions of each account;
# only accounts in use unless unused.
#! no_system_account: true
#! accounts {
#!   A {
#!     users [{user: a, password: p}]
#!     exports [{service: svc.>}, {stream: feed.>}]
#!   }
#!   B {
#!     users [{user: b, password: p}]
#!     imports [
#!       {service: {account: A, subject: svc.x}}
#!       {stream: {account: A, subject: feed.>}}
#!     ]
#!   }
#!   C { users [{user: c, password: p}] }
#! }
#= account_statz
# Go counts what the clients of every account publish in $G, the account
# they are in before CONNECT, and what they get in the account it was
# published in; naq both in the account of the client.
#- received
#- sent
# The harness asks /connz until a closed client shows.
#~ /connz
