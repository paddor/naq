# $SYS.DEBUG.SUBSCRIBERS counts the subscribers of a subject, or of a
# queue group on it, in the account that asks: through an import of the
# system account's export, or in the system account itself. A filter
# with wildcards counts the subscriptions within it. nats-server takes the
# import only of an export in the config.
#! server_name: s1
#! accounts {
#!   SYS {
#!     users [{user: sys, password: p}]
#!     exports [{service: "$SYS.DEBUG.SUBSCRIBERS"}]
#!   }
#!   A {
#!     users [{user: a, password: p}]
#!     imports [{service: {account: SYS, subject: "$SYS.DEBUG.SUBSCRIBERS"}}]
#!   }
#! }
#! system_account: SYS
