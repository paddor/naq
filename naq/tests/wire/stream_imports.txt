# A stream import delivers the exporter's messages in the importer, with
# the subjects rewritten.
#! accounts {
#!   E {
#!     users: [{user: e, password: p}]
#!     exports: [{stream: "s.>"}, {stream: lit}, {stream: "w.*"}, {stream: "self.>"}]
#!     imports: [{stream: {account: E, subject: "self.>"}, prefix: me}]
#!   }
#!   I {
#!     users: [{user: i, password: p}]
#!     imports: [
#!       {stream: {account: E, subject: "s.>"}, prefix: pre}
#!       {stream: {account: E, subject: lit}, to: "x.*"}
#!       {stream: {account: E, subject: "w.*"}, to: "ww.$1.end"}
#!       {stream: {account: E, subject: "s.d"}, to: "pre.s.d"}
#!     ]
#!   }
#! }

e> CONNECT {"verbose":false,"user":"e","pass":"p"}
e> PING
e< INFO {"auth_required":true,"client_ip":"127.0.0.1","headers":true,"max_payload":1048576,"proto":1}
e< PONG
i> CONNECT {"verbose":false,"user":"i","pass":"p"}
i> SUB pre.s.> 1
i> SUB pre.*.x 3
i> SUB x.y 4
i> SUB ww.*.end 5
i> PING
i< INFO {"auth_required":true,"client_ip":"127.0.0.1","headers":true,"max_payload":1048576,"proto":1}
i< PONG
e> PUB s.a 1
