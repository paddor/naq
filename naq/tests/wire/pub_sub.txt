# Two clients, a reply subject, UNSUB.
a> CONNECT {"verbose":false}
a> SUB foo 1
a> SUB foo 2
a< INFO {"client_ip":"127.0.0.1","headers":true,"max_payload":1048576,"proto":1}
b> CONNECT {"verbose":false}
b> PUB foo bar.reply 5
b> hello
b< INFO {"client_ip":"127.0.0.1","headers":true,"max_payload":1048576,"proto":1}
a< MSG foo 1 bar.reply 5
a< hello
a< MSG foo 2 bar.reply 5
a< hello
a> UNSUB 1
b> PUB foo 3
b> two
a< MSG foo 2 3
a< two
a> UNSUB 2
b> PUB foo 5
b> gone!
